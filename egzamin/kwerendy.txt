SELECT nazwa FROM `obiekty` WHERE panstwo = "Islandia" and idRodzaj = 10;

SELECT rodzaje.rodzaj, COUNT(obiekty.idObiekt) as Liczba_obiektow FROM `rodzaje` inner join obiekty on rodzaje.idRodzaj = obiekty.idRodzaj where rodzaje.rodzaj = "wodospad" or rodzaje.rodzaj = "szczyt" GROUP BY rodzaje.rodzaj;
